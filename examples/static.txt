using System;


public class Program
{
    public static void Main()
    {

        //static and instance example
        Circle circle = new Circle(5);
        Console.WriteLine($"Area of the circle: {circle.GetArea()}");

    }
}

public class Circle
{
    static float pi;
    int radius;
    static Circle()
    {
        Circle.pi = 3.14f;
    }

    public Circle(int r)
    {
        this.radius = r;
    }
    public float GetArea()
    {
        return Circle.pi * radius * radius;
    }
}
